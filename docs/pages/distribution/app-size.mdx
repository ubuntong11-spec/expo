async function loadCardNos(){
  if(!fb)return;
  try{
    const[s1,s2]=await Promise.all([db.collection('stock_in').orderBy('cardNo').get(),db.collection('stock_out').orderBy('cardNo').get()]);
    allInDocs=s1.docs.map(d=>({id:d.id,...d.data()}));
    allOutDocs=s2.docs.map(d=>({id:d.id,...d.data()}));
    const inNos=[...new Set(allInDocs.filter(d=>docYear(d)===selectedYear).map(d=>d.cardNo))].sort();

}
